from langgraph.graph import StateGraph
from state import AgentState

def build_graph():
    workflow = StateGraph(AgentState)

    # Register seven specialized agents
    workflow.add_node("ioc",        extract_iocs)      # IOC Extractor
    workflow.add_node("threat",     threat_analysis)   # Rule-based scorer
    workflow.add_node("virustotal", virustotal_agent)  # Threat Intel API
    workflow.add_node("memory",     memory_agent)      # Pattern memory
    workflow.add_node("reasoning",  reasoning_agent)   # LLM classification
    workflow.add_node("report",     reporting_agent)   # Report writer

    # Linear pipeline: each agent enriches shared state
    workflow.set_entry_point("ioc")
    workflow.add_edge("ioc",        "threat")
    workflow.add_edge("threat",     "virustotal")
    workflow.add_edge("virustotal", "memory")
    workflow.add_edge("memory",     "reasoning")
    workflow.add_edge("reasoning",  "report")

    return workflow.compile()
